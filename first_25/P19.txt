    SELECT "query_name" , ROUND(AVG(("rating" * 1.0) / ("position" * 1.0)) , 2) AS "quality"
    FROM "Queries"
    GROUP BY "query_name"
) , e AS (
SELECT "query_name" , COUNT(*) AS "total"
FROM "Queries" 
 GROUP BY "query_name"

), d AS (
    SELECT "e"."query_name" , ROUND (COUNT("rating") * 100.0 / "total" , 2) AS "poor_query_percentage"
    FROM "Queries" AS "q" RIGHT JOIN "e"
    ON "q"."query_name" = "e"."query_name"
    WHERE "rating" < 3
    GROUP BY "e"."query_name" , "total"
) SELECT "c"."query_name" , "quality" , ISNULL("poor_query_percentage" , 0) AS "poor_query_percentage"
