WITH c AS (
SELECT "p"."product_id" , "price" ,"units" , ("price" * "units") AS "total"
FROM "Prices" AS "p" LEFT JOIN "UnitsSold" AS "u"
ON "p"."product_id" = "u"."product_id"
WHERE ("purchase_date" BETWEEN "start_date" AND "end_date") OR "purchase_date" IS NULL 
)  , e AS (
    SELECT "product_id" , SUM("total") AS "Tprice" , SUM("units") AS "Tunits"
    FROM c 
    GROUP BY "product_id" 
)
   SELECT "product_id" , ISNULL(ROUND(("Tprice" * 1.0/ "Tunits" *1.0) ,2),0)
    AS "average_price"
    FROM e ;
