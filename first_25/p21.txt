WITH c AS (
SELECT "customer_id" , LAG("customer_id") OVER(ORDER BY "customer_id" , "order_date" ) AS "prev" , "order_date" ,
"customer_pref_delivery_date"
FROM "Delivery" 
) , d AS (
    SELECT (CASE WHEN "order_date" = "customer_pref_delivery_date" THEN 'immediate' 
    ELSE 'scheduled' END ) AS "type" FROM "c"
    WHERE "customer_id" != "prev" OR "prev" IS NULL   
) SELECT ROUND(SUM(CASE WHEN "type" = 'immediate' THEN 1 ELSE 0 END) * 100.0 / COUNT("type"),2)
 AS "immediate_percentage"
 FROM d ;
